find_package(PkgConfig REQUIRED)
pkg_check_modules(GTK4 REQUIRED IMPORTED_TARGET gtk4)

add_executable(mewnet-gtk main.c)
target_include_directories(mewnet-gtk PRIVATE "${MEWNET_PACKAGE_DIR}")
target_link_libraries(mewnet-gtk PRIVATE "${MEWNET_LIBRARY}" PkgConfig::GTK4)
target_compile_options(mewnet-gtk PRIVATE -Wall -Wextra)
if(CMAKE_C_COMPILER_ID MATCHES "Clang")
    target_link_options(mewnet-gtk PRIVATE "-Wl,--allow-shlib-undefined")
endif()
set_target_properties(mewnet-gtk PROPERTIES BUILD_RPATH "${MEWNET_ROOT}/dist")
